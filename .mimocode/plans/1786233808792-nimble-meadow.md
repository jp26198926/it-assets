# Fix Stock Movements Advanced Search + Default Date Filter

## Problem

1. **Advanced Search not visible** — The toolbar conditionally renders the Advanced Search button only when `onServerSearch` is provided, but the page doesn't pass it.
2. **Advanced Search should be server-side** — Currently uses client-side filtering; needs to query the database directly.
3. **Default load should show current date only** — Page currently loads all records; should default to today's date.
4. **Wrong advanced search fields** — Needs: Date From, Date To, Transaction Type, Item, Location, Reference, Remarks.

### 2. `lib/services/stock-movement-service.ts` — Add server-side filter support

Update `getStockMovements` to accept `StockMovementFilters` and build MongoDB queries:

```ts
export async function getStockMovements(filters?: StockMovementFilters): Promise<StockMovement[]> {
  await connectDB();

  const query: Record<string, unknown> = {};

  // Default: current date (start of day to end of day)
  if (!filters || (!filters.date_from && !filters.date_to && !filters.transaction_type && !filters.item_name && !filters.location_name && !filters.reference_description && !filters.remarks)) {
    const now = new Date();
    const startOfDay = new Date(now.getFullYear(), now.getMonth(), now.getDate());
    const endOfDay = new Date(now.getFullYear(), now.getMonth(), now.getDate() + 1);
    query.date = { $gte: startOfDay, $lt: endOfDay };
  } else {
    // Date range filter
    if (filters.date_from || filters.date_to) {
      const dateQuery: Record<string, Date> = {};
      if (filters.date_from) dateQuery.$gte = new Date(filters.date_from);
      if (filters.date_to) {
        const end = new Date(filters.date_to);
        end.setDate(end.getDate() + 1);
        dateQuery.$lt = end;
      }
      query.date = dateQuery;
    }
  }

  // Transaction type (exact match)
  if (filters?.transaction_type) {
    query.transaction_type = filters.transaction_type;
  }

  // Item name (regex search on populated field — need to find item IDs first)
  if (filters?.item_name) {
    const { Item } = await import("@/lib/db/models/item");
    const items = await Item.find({ name: { $regex: filters.item_name, $options: "i" } }).select("_id").lean();
    query.item_id = { $in: items.map(i => i._id) };
  }

  // Location name (regex search on populated field — need to find location IDs first)
  if (filters?.location_name) {
    const { Location } = await import("@/lib/db/models/location");
    const locations = await Location.find({ name: { $regex: filters.location_name, $options: "i" } }).select("_id").lean();
    query.location_id = { $in: locations.map(l => l._id) };
  }

  // Reference description (regex)
  if (filters?.reference_description) {
    query.reference_description = { $regex: filters.reference_description, $options: "i" };
  }

  // Remarks (regex)
  if (filters?.remarks) {
    query.remarks = { $regex: filters.remarks, $options: "i" };
  }

  // ... rest of query execution
}
```
